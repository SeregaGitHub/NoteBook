    Разбиение массива на 2 партиции (для быстрой сортировки)

private static int partition(int[] data, int leftIndex, int rightIndex) {
        // int pivot = data[(leftIndex + rightIndex) / 2];
        int middle = data[(leftIndex + rightIndex) / 2];
        int pivot = Math.max(Math.min(data[leftIndex], middle), Math.min(Math.max(data[leftIndex], middle), data[rightIndex]));
        while (leftIndex <= rightIndex) {
            while (data[leftIndex] < pivot) {
                leftIndex++;
            }
            while (data[rightIndex] > pivot) {
                rightIndex--;
            }
            if (leftIndex <= rightIndex) {
                swap(data, leftIndex, rightIndex);
                leftIndex++;
                rightIndex--;
            }
        }
        return leftIndex;
    }

import java.io.InputStreamReader;
import java.io.BufferedReader;
import java.io.IOException;
import java.lang.StringBuilder;
import java.util.StringTokenizer;

class Solution {
    public static void main(String[] args) throws IOException {
        StringBuilder output_buffer = new StringBuilder();
        BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
        int num_lines = Integer.parseInt(reader.readLine());
        for (int i = 0; i < num_lines; ++i) {
            StringTokenizer tokenizer = new StringTokenizer(reader.readLine());
            int value_1 = Integer.parseInt(tokenizer.nextToken());
            int value_2 = Integer.parseInt(tokenizer.nextToken());
            int result = value_1 + value_2;
            output_buffer.append(result).append("\n");
        }
        System.out.println(output_buffer.toString());
    }
}
